@main struct HarnessApp: App {
    init() {
        if let shot = ProcessInfo.processInfo.environment["MAGZI_TAB"] {
            let map: [String: MagziRouter.Tab] = ["home": .home, "magzi": .magzi, "household": .household, "adhoc": .adhoc, "more": .more]
            MagziRouter.shared.tab = map[shot] ?? .home
        }
    }
    var body: some Scene { WindowGroup { HarnessRoot().task {
        await MagziHub.shared.start()
        if ProcessInfo.processInfo.environment["MAGZI_MOCK"] == "1", !MagziAuthStore.shared.isPaired {
            MagziAuthStore.shared.serverURLString = "http://127.0.0.1:8767"
            _ = try? await MagziAPI.shared.verifyLogin(email: "owner@example.com", code: "000000")
            await MagziHub.shared.refresh()
        }
        if let text = ProcessInfo.processInfo.environment["MAGZI_AUTOSEND"] {
            try? await Task.sleep(nanoseconds: 2_000_000_000)
            MagziRouter.shared.askMagzi(text, send: true)
        }
    } } }
}

        switch ProcessInfo.processInfo.environment["MAGZI_SCREEN"] {

// MARK: Ad-hoc bridge harness — the real AIChatViewModel+MagziBridge.swift runs against these.
struct AgentToolDefinition {
    let name: String
    init(name: String, description: String, parameters: [String: AgentToolParam], required: [String], propertyOrdering: [String]? = nil) { self.name = name }
}
struct AgentToolParam { init(type: AgentParamType, description: String, enumValues: [String]? = nil) {} }
enum AgentParamType: String { case string, integer, boolean, number, array, object }
@MainActor final class AIChatViewModel {
    struct FileToolResult { let output: String; let success: Bool }
}

struct BridgeHarnessView: View {
    @State private var log = "running bridge tools…"
    var body: some View {
        ScrollView { Text(log).font(.system(.footnote, design: .monospaced)).padding() }
            .task {
                let vm = AIChatViewModel()
                var out = "tools offered: \(vm.makeMagziBridgeTools().map(\.name))\n\n"
                let h = await vm.executeMagziBridgeTool(name: "magzi_household", argsJson: #"{"view":"bills"}"#)
                out += "magzi_household(bills) ok=\(h.success)\n\(h.output.prefix(300))\n\n"
                log = out + "magzi_ask… (waiting for Magzi)"
                let a = await vm.executeMagziBridgeTool(name: "magzi_ask", argsJson: #"{"request":"Add bin bags to the shopping list, and tell me what is planned for dinner on Tuesday."}"#)
                out += "magzi_ask ok=\(a.success)\n\(a.output)"
                log = out
                print("BRIDGE-RESULT\n" + out)
            }
    }
}
